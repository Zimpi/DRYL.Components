# DrylAiField Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `DrylAiField` — a wrapper that gives any existing text-like DRYL input a ✨ AI affordance (generate / transform selection, live streaming into the field, accept/reject) with zero wiring beyond passing an `AIAgent`.

**Architecture:** The wrapper broadcasts `AiState` to the wrapped input via `DrylAiScope` (aura for free, no touching existing components), runs the agent through the existing `DrylAgentRunner.Start(...)`, and reads/writes the field's value through a new tiny ES module (`dryl-aifield.js`) that dispatches native `input` events so the inner `@bind-Value` updates itself. Mini-prompt lives in a `DrylPopover` (body portal, Esc/click-outside built in). Spec: `docs/superpowers/specs/2026-07-10-ai-field-design.md`.

**Tech Stack:** Blazor RCL (net8/9/10), Microsoft.Agents.AI 1.13.0 (`ChatClientAgent`, `AgentSession`), bunit (`BunitContext`, `JSInterop.SetupModule`), xUnit.

using System.Text;

namespace DRYL.Components.Agents;

/// <summary>
/// Builds the English model prompt for a <see cref="DrylAiField"/> invocation and cleans the reply.
/// English template on purpose — models follow the "respond with ONLY the replacement" contract
/// more reliably; the user-facing UI strings stay German.
/// </summary>
internal static class AiFieldPrompt
{
    /// <summary>Composes instruction + optional context + current value + optional selection.</summary>
    internal static string Build(string instruction, string? context, string value, string? selection)
    {
        var sb = new StringBuilder();
        sb.AppendLine(instruction.Trim());

        if (!string.IsNullOrWhiteSpace(context))
        {
            sb.AppendLine();
            sb.AppendLine("Additional context:");
            sb.AppendLine(context.Trim());
        }

        if (!string.IsNullOrEmpty(value))
        {
            sb.AppendLine();
            sb.AppendLine("Current field value:");
            sb.AppendLine("\"\"\"");
            sb.AppendLine(value);
            sb.AppendLine("\"\"\"");
        }

        if (selection is not null)
        {
            sb.AppendLine();
            sb.AppendLine("Selected portion (transform ONLY this):");
            sb.AppendLine("\"\"\"");
            sb.AppendLine(selection);
            sb.AppendLine("\"\"\"");
        }

        sb.AppendLine();
        sb.Append("Respond with ONLY the replacement text for ");
        sb.Append(selection is not null ? "the selected portion" : "the field");
        sb.Append(" — no quotes, no markdown fences, no explanation.");
        return sb.ToString();
    }

    /// <summary>
    /// Defensive post-trim: strips one wrapping pair of double quotes or one fenced code block
    /// if the model added them despite the contract.
    /// </summary>
    internal static string Clean(string raw)
    {
        var text = raw.Trim();

        if (text.StartsWith("```", StringComparison.Ordinal))
        {
            var firstBreak = text.IndexOf('\n');
            var lastFence = text.LastIndexOf("```", StringComparison.Ordinal);
            if (firstBreak >= 0 && lastFence > firstBreak)
                return text[(firstBreak + 1)..lastFence].Trim();
        }
        else if (text.Length >= 2 && text[0] == '"' && text[^1] == '"')
        {
            return text[1..^1];
        }

        return text;
    }
}
```

namespace DRYL.Components.Agents;

/// <summary>DOM snapshot of the wrapped field, returned by the <c>dryl-aifield.js</c> module.</summary>
internal sealed record AiFieldSnapshot
{
    /// <summary>Whether a text-like input/textarea was found inside the wrapper.</summary>
    public bool Found { get; init; }

    /// <summary>The field's current value ("" when not found).</summary>
    public string Value { get; init; } = string.Empty;

    /// <summary>Selection start, or -1 when the element exposes no selection API.</summary>
    public int SelStart { get; init; } = -1;

    /// <summary>Selection end, or -1 when the element exposes no selection API.</summary>
    public int SelEnd { get; init; } = -1;

    /// <summary>True when a non-empty text range is selected.</summary>
    public bool HasSelection => Found && SelStart >= 0 && SelEnd > SelStart;
}

```js
// dryl-aifield.js — DOM value bridge for DrylAiField (lazy ES module, no consumer wiring).
// Finds the first text-like field inside the wrapper, reads value + selection, and writes
// streamed text back by dispatching a native bubbling `input` event so the wrapped
// component's @bind-Value updates itself. Skips anything inside [data-aifield-ui]
// (the wrapper's own trigger / prompt UI).

const SELECTOR =
    'textarea, input:not([type]), input[type="text"], input[type="search"], ' +
    'input[type="email"], input[type="url"], input[type="tel"]';

function field(root) {
    if (!root) return null;
    for (const el of root.querySelectorAll(SELECTOR)) {
        if (el.disabled) continue;
        if (el.closest('[data-aifield-ui]')) continue;
        return el;
    }
    return null;
}

export function snapshot(root) {
    const el = field(root);
    if (!el) return { found: false, value: '', selStart: -1, selEnd: -1 };
    let selStart = -1, selEnd = -1;
    try {
        selStart = el.selectionStart ?? -1;
        selEnd = el.selectionEnd ?? -1;
    } catch { /* some input types throw on selection access */ }
    return { found: true, value: el.value ?? '', selStart, selEnd };
}

export function write(root, text) {
    const el = field(root);
    if (!el) return;
    el.value = text;
    el.dispatchEvent(new Event('input', { bubbles: true }));
}

export function setBusy(root, busy) {
    const el = field(root);
    if (el) el.readOnly = !!busy;
}

export function focusField(root) {
    const el = field(root);
    if (el) el.focus({ preventScroll: false });
}
```

```csharp
    private async Task HandleTriggerClickAsync()
    {
        switch (_phase)
        {
            case FieldPhase.Running:
                CancelRun();
                break;
            case FieldPhase.Idle when UsesPrompt:
                _promptText = ShowPrompt ? (Instruction ?? string.Empty) : string.Empty;
                _promptOpen = true;
                break;
            case FieldPhase.Idle:
                await StartAsync(Instruction!);
                break;
            // Review: the chips are the affordance — the trigger is inert.
        }
    }

    private void CancelRun() => _cts?.Cancel();

    private async Task StartAsync(string instruction)
    {
        if (_phase == FieldPhase.Running) return;
        _errorText = null;

        var module = await GetModuleAsync();
        var snap = await module.InvokeAsync<AiFieldSnapshot>("snapshot", _root);
        if (!snap.Found) return;   // nothing text-like inside — the affordance is a no-op

        _snapshotValue = snap.Value;
        string? selection = null;
        _prefix = _suffix = string.Empty;
        if (snap.HasSelection)
        {
            selection = snap.Value[snap.SelStart..snap.SelEnd];
            _prefix = snap.Value[..snap.SelStart];
            _suffix = snap.Value[snap.SelEnd..];
        }

        var prompt = AiFieldPrompt.Build(instruction, Context, snap.Value, selection);

        _cts?.Cancel();
        _cts?.Dispose();
        _cts = new CancellationTokenSource();
        var ct = _cts.Token;

        _phase = FieldPhase.Running;
        SetAura(AiState.Thinking);
        Announce("Die AI schreibt …");
        await module.InvokeVoidAsync("setBusy", _root, true);

        try
        {
            var session = SessionFactory is not null
                ? await SessionFactory(ct)
                : await Agent.CreateSessionAsync(ct);

            _run = Runner.Start(Agent, session, prompt, aiKey: null, ct);
            await ConsumeAsync(_run, module, ct);
        }
        catch (OperationCanceledException)
        {
            await RestoreAsync("Vorschlag verworfen.");
            await OnRejected.InvokeAsync();
        }
        catch (Exception ex)
        {
            await FailAsync(new DrylRunError(ex.Message, ex));
        }
    }

    private async Task ConsumeAsync(DrylAgentRun run, IJSObjectReference module, CancellationToken ct)
    {
        var sb = new StringBuilder();
        var lastFlush = Environment.TickCount64 - FlushMs;   // first delta flushes immediately

        await foreach (var delta in run.TextStream.WithCancellation(ct))
        {
            sb.Append(delta);
            if (_aura != AiState.Streaming) SetAura(AiState.Streaming);
            if (Environment.TickCount64 - lastFlush >= FlushMs)
            {
                lastFlush = Environment.TickCount64;
                await module.InvokeVoidAsync("write", ct, _root, Compose(sb.ToString()));
            }
        }

        await run.WaitForCompletionAsync();
        ct.ThrowIfCancellationRequested();

        if (run.Error is not null)
        {
            await FailAsync(run.Error);
            return;
        }

        _finalText = Compose(AiFieldPrompt.Clean(sb.ToString()));
        await module.InvokeVoidAsync("write", ct, _root, _finalText);
        await module.InvokeVoidAsync("setBusy", ct, _root, false);

        _phase = FieldPhase.Review;
        SetAura(AiState.Generated);
        Announce("AI-Vorschlag bereit — übernehmen oder verwerfen.");
        await InvokeAsync(StateHasChanged);
    }

    private string Compose(string core) => _prefix + core + _suffix;

    private async Task RestoreAsync(string announcement)
    {
        _phase = FieldPhase.Idle;
        SetAura(AiState.None);
        Announce(announcement);
        try
        {
            var module = await GetModuleAsync();
            await module.InvokeVoidAsync("write", _root, _snapshotValue);
            await module.InvokeVoidAsync("setBusy", _root, false);
            await module.InvokeVoidAsync("focusField", _root);
        }
        catch (JSDisconnectedException) { /* circuit gone */ }
        await InvokeAsync(StateHasChanged);
    }

    private async Task FailAsync(DrylRunError error)
    {
        _errorText = error.Message;
        await RestoreAsync("Die AI-Anfrage ist fehlgeschlagen.");
        await OnError.InvokeAsync(error);
    }

    private void SetAura(AiState state)
    {
        _aura = state;
        _ = InvokeAsync(StateHasChanged);
    }

    private void Announce(string message)
    {
        _live = message;
        _ = InvokeAsync(StateHasChanged);
    }
